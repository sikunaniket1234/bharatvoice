# Machine-learning extras for BharatVoice AI inference.
#
# Kept separate from requirements.txt so the API contract, health endpoints and
# unit tests keep running on machines that have no ML stack installed. Without
# these packages the service starts normally and reports itself not-ready.
#
# IMPORTANT: this stack is Linux/macOS only. IndicTransToolkit ships
# IndicProcessor only as a compiled Cython extension and publishes no
# win_amd64 wheel for any Python version, so it cannot be installed on Windows
# x86-64 without first installing the Microsoft C++ Build Tools and compiling
# it from the sdist. A cp312 manylinux_x86_64 wheel does exist, so the intended
# and supported path is the Docker `ai` service. The Windows virtualenv in this
# repository is deliberately kept contract-only.
#
# The CUDA build is selected by the installer, not here:
#   pip install --extra-index-url https://download.pytorch.org/whl/cu128 -r requirements-ml.txt
#
# transformers is pinned to 4.46.x. It is not merely bounded below 5.0.0: the
# IndicTrans2 model cards state that the "translation" pipeline was removed in
# transformers v5, and these checkpoints are trust_remote_code models whose
# bundled modeling code is tightly coupled to one generation of the 4.x API.
# Specifically, modeling_indictrans.py does
#   past_key_values[0][0].shape[2] if past_key_values is not None else 0
# which raises AttributeError under the Cache object introduced later in 4.x,
# because a fresh DynamicCache returns None for index 0. Verified on this
# hardware: 4.57.6 fails with
#   AttributeError: 'NoneType' object has no attribute 'shape'
# and 4.46.3 generates correctly. Do not widen this range without re-running the
# en->or and or->en benchmark.
#
# flash_attention_2 is intentionally NOT enabled. The model cards say not to
# set it when flash-attn is unavailable, and it has no supported Windows build.

torch>=2.6,<3.0
transformers>=4.46,<4.47
sentencepiece>=0.2.0
sacremoses>=0.1.1
accelerate>=0.34
huggingface_hub>=0.27

# Required by the IndicTrans2 model cards for normalisation and postprocessing:
#   from IndicTransToolkit.processor import IndicProcessor
# Linux/macOS wheels only. See the note above.
IndicTransToolkit==1.1.1

# Evaluation metrics for the controlled bilingual benchmark set.
sacrebleu>=2.4.0

# Optional, but the benchmark script reports process RSS when it is present.
psutil>=5.9
